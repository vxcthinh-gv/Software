	\loigiai{
		{\color{purple}\textbf{Cách 1.}}\\

				\begin{cases}u_{4}&=10\\ u_{4}+u_{6}&=26\end{cases} &\Leftrightarrow \begin{cases}u_1+3d&=10\\ (u_1+3d)+(u_1+5d)&=26\end{cases} \\

			\end{align*}

			\begin{align*}
				\Leftrightarrow \begin{cases}u_1&=1\\ d&=3\end{cases}
			\end{align*}
			Vậy công sai của cấp số cộng là $d=3$.

		\noindent
		{\color{purple}\textbf{Cách 2.}}\\
		Thế giá trị của $u_4$ vào phương trình thứ hai của hệ, ta có:

			$$u_6 - u_4 = 2d \Leftrightarrow 16 - 10 = 2d \Leftrightarrow 2d = 6 \Leftrightarrow d = 3$$

		\noindent
		{\color{purple}\textbf{Cách 3.}}\\
		Áp dụng tính chất của cấp số cộng $u_4 + u_6 = 2u_5$, ta suy ra:
			$$2u_5 = 26 \Leftrightarrow u_5 = 13$$
		 Theo định nghĩa cấp số cộng, công sai được tính bằng:
			$$d = u_5 - u_4 = 13 - 10 = 3$$
		Vậy công sai của cấp số cộng là $d=3$.		
	}

\begin{ex}%[1D6H4-2]
	Tổng bình phương các nghiệm của phương trình $\log_{2}(x^{2}-5x+7)=0$ bằng
	\choice
	{$6$}
	{$7$}
	{\True $13$}
	{$5$}
	\loigiai{
		\textbf{\color{purple} Cách 1.}\\
		 Biến đổi phương trình đã cho, ta được:
			\begin{align*}
				\log_2(x^2-5x+7) = 0 &\Leftrightarrow x^2 - 5x + 7 = 2^0 \\
				&\Leftrightarrow x^2 - 5x + 7 = 1 \\
				&\Leftrightarrow x^2 - 5x + 6 = 0 \\
				&\Leftrightarrow \left[\begin{aligned} &x = 2 \\ &x = 3 \end{aligned}\right.
			\end{align*}
		Vậy phương trình có hai nghiệm là $x=2$ và $x=3$.
		Tổng bình phương các nghiệm của phương trình là $2^2 + 3^2 = 4 + 9 = 13$.\\
		\noindent
		\textbf{\color{purple} Cách 2.}\\
		Ta có:
			$$ \log_2(x^2-5x+7) = 0 \Leftrightarrow x^2 - 5x + 6 = 0 \quad (1) $$
		 Phương trình $(1)$ có $\Delta = (-5)^2 - 4 \cdot 1 \cdot 6 = 1 > 0$ nên luôn có hai nghiệm phân biệt $x_1, x_2$.\\
		Áp dụng định lí Vi-ét, ta có:
			$$ \begin{cases} x_1 + x_2 = 5 \\ x_1x_2 = 6 \end{cases} $$
		Tổng bình phương các nghiệm của phương trình là:
			$$ x_1^2 + x_2^2 = (x_1 + x_2)^2 - 2x_1x_2 = 5^2 - 2 \cdot 6 = 25 - 12 = 13. $$
	}
\end{ex}

		\textbf{\color{purple} Cách 1.}\\
		Gọi $E, F$ lần lượt là trung điểm của $AB, CD$.\\
		Vì $M, N$ lần lượt là trọng tâm của các tam giác $SAB$ và $SCD$ nên $M \in SE$, $N \in SF$ và ta có tỉ số:
		$$ \dfrac{SM}{SE} = \dfrac{2}{3}, \quad \dfrac{SN}{SF} = \dfrac{2}{3} \Rightarrow \dfrac{SM}{SE} = \dfrac{SN}{SF} $$
		Xét tam giác $SEF$, có $\dfrac{SM}{SE} = \dfrac{SN}{SF}$ nên theo định lí Thales đảo ta có $MN \parallel EF$.\\

		\noindent
		\textbf{\color{purple} Cách 2.}\\
		Gọi $E, F$ lần lượt là trung điểm của $AB, CD$.\\
		Ta phân tích vectơ $\vec{MN}$ theo quy tắc ba điểm:
			\begin{align*}
				\vec{MN} &= \vec{MS} + \vec{SN} \\
				&= -\dfrac{2}{3}\vec{SE} + \dfrac{2}{3}\vec{SF} \\
				&= \dfrac{2}{3}(\vec{SF} - \vec{SE}) \\
				&= \dfrac{2}{3}\vec{EF}
			\end{align*}
		Do $\vec{MN} = \dfrac{2}{3}\vec{EF}$ nên giá của hai vectơ này song song với nhau (do điểm $M$ không thuộc đường thẳng $EF$), tức là $MN \parallel EF$.\\
		 Mà đường thẳng $EF$ nằm trong mặt phẳng $(ABCD)$ và $MN \not\subset (ABCD)$.\\
		Vậy $MN \parallel (ABCD)$.
	}
\end{ex}

	\shortans{4.33}
	\loigiai{
		Căn phòng là hình hộp chữ nhật $ABCD.A'B'C'D'$. Theo giả thiết ta có $AB=CD=A'B'=C'D'=3$ m; $BC=AD=B'C'=A'D'=4$ m và chiều cao $AA'=BB'=CC'=DD'=4$ m.\\
		{\color{blue} Phân tích chuyển động của hai con kiến}\\
		Quỹ đạo kiến 1 là $A' \to B \to C$. Độ dài đoạn đầu tiên là $A'B = \sqrt{AA'^2 + AB^2} = \sqrt{4^2 + 3^2} = 5$ m.\\
		Quỹ đạo kiến 2 là $D \to C' \to C$. Độ dài đoạn đầu tiên là $DC' = \sqrt{DD'^2 + C'D'^2} = \sqrt{4^2 + 3^2} = 5$ m.\\
		Tổ kiến ở $C$. Khi kiến 2 còn cách tổ $1$ m, nghĩa là nó đang trên đoạn $C'C$ và cách $C$ đoạn $1$ m.\\
		Quãng đường kiến 2 đã đi là: $s_2 = DC' + (C'C - 1) = 5 + (4 - 1) = 8$ m.\\
		Thời gian di chuyển của kiến 2 (cũng là của kiến 1 vì xuất phát cùng lúc) là: $t = \dfrac{s_2}{v_2} = \dfrac{8}{0,06} = \dfrac{400}{3}$ (s).\\
		Trong thời gian $t$, quãng đường kiến 1 bò được là: $s_1 = v_1 \cdot t = 0,02 \cdot \dfrac{400}{3} = \dfrac{8}{3}$ m.\\
		Vì $s_1 = \dfrac{8}{3} < A'B = 5$ nên lúc này kiến 1 vẫn đang nằm trên đoạn $A'B$.\\
		{\color{blue} Gắn hệ trục tọa độ và tính toán}
		Chọn hệ trục tọa độ $Axyz$ sao cho gốc $A(0;0;0)$, các tia $AD, AB, AA'$ lần lượt trùng với các tia $Ox, Oy, Oz$.\\
		Khi đó ta có tọa độ các đỉnh: $A(0;0;0)$, $D(4;0;0)$, $B(0;3;0)$, $C(4;3;0)$, $A'(0;0;4)$, $C'(4;3;4)$.\\
		Gọi $M$ là vị trí của kiến 1. Điểm $M \in A'B$ sao cho $A'M = \dfrac{8}{3}$ m.\\
		Suy ra $\overrightarrow{A'M} = \dfrac{A'M}{A'B} \overrightarrow{A'B} = \dfrac{8/3}{5} \overrightarrow{A'B} = \dfrac{8}{15} \overrightarrow{A'B}$.\\
		Ta có $\overrightarrow{A'B} = (0; 3; -4) \Rightarrow \overrightarrow{A'M} = \left(0; \dfrac{24}{15}; -\dfrac{32}{15}\right) = \left(0; \dfrac{8}{5}; -\dfrac{32}{15}\right)$.\\
		Tọa độ điểm $M$: $\begin{cases} x_M = 0 + 0 = 0 \\ y_M = 0 + \dfrac{8}{5} = \dfrac{8}{5} \\ z_M = 4 - \dfrac{32}{15} = \dfrac{28}{15} \end{cases} \Rightarrow M\left(0; \dfrac{8}{5}; \dfrac{28}{15}\right)$.\\
		Gọi $N$ là vị trí của kiến 2. Điểm $N \in C'C$ và cách $C(4;3;0)$ một khoảng $1$ m. \\
		Do đường thẳng $C'C$ song song với trục $Oz$ và tọa độ $z$ giảm dần từ $4$ xuống $0$, nên tọa độ $N(4; 3; 1)$.\\
		Khoảng cách giữa hai con kiến lúc này là $MN$:
		$$MN = \sqrt{(4-0)^2 + \left(3 - \dfrac{8}{5}\right)^2 + \left(1 - \dfrac{28}{15}\right)^2} = \sqrt{16 + \left(\dfrac{7}{5}\right)^2 + \left(-\dfrac{13}{15}\right)^2}$$
		$$MN = \sqrt{16 + \dfrac{49}{25} + \dfrac{169}{225}} = \sqrt{\dfrac{3600 + 441 + 169}{225}} = \sqrt{\dfrac{4210}{225}} = \dfrac{\sqrt{4210}}{15} \approx 4,33.$$

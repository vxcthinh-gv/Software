	{\True Khảo sát ngẫu nhiên $10$ khách hàng, xác suất để có đúng $3$ người sử dụng cả hai dịch vụ xấp xỉ bằng $0{,}27$}
	{Nghiên cứu thêm về tính năng \lq\lq Hoàn tiền\rq\rq (Cashback), ngân hàng nhận thấy: tỷ lệ khách hàng được hoàn tiền nếu sử dụng cả hai dịch vụ là $60\%$, còn với nhóm khách hàng không sử dụng đồng thời cả hai dịch vụ này thì tỷ lệ được hoàn tiền chỉ là $10\%$. Biết rằng một khách hàng vừa nhận được tin nhắn hoàn tiền từ hệ thống, xác suất để khách hàng đó thuộc nhóm sử dụng cả hai dịch vụ là $0{,}25$}
	\loigiai{

		$A$: \lq\lq Khách hàng sử dụng dịch vụ thanh toán hóa đơn bằng mã QR\rq\rq. $\mathrm{P}(A) = 0{,}7$. \\
		$B$: \lq\lq Khách hàng sử dụng dịch vụ gửi tiết kiệm trực tuyến\rq\rq. $\mathrm{P}(B) = 0{,}5$. \\
		$A \cap B$: \lq\lq Khách hàng sử dụng cả hai dịch vụ\rq\rq. $\mathrm{P}(A \cap B) = 0{,}3$.

			\itemch Xác suất để một khách hàng sử dụng ít nhất một trong hai dịch vụ là
			\[\mathrm{P}(A \cup B) = \mathrm{P}(A) + \mathrm{P}(B) - \mathrm{P}(A \cap B) = 0{,}7 + 0{,}5 - 0{,}3 = 0{,}9.\]
			\itemch Xác suất khách hàng sử dụng dịch vụ thanh toán bằng mã QR với điều kiện đã dùng dịch vụ gửi tiết kiệm trực tuyến là
			\[\mathrm{P}(A \mid B) = \dfrac{\mathrm{P}(A \cap B)}{\mathrm{P}(B)} = \dfrac{0{,}3}{0{,}5} = 0{,}6.\]
			\itemch Gọi $X$ là số khách hàng sử dụng cả hai dịch vụ trong nhóm $10$ người được chọn. Vì các khách hàng được chọn ngẫu nhiên và độc lập nên $X$ tuân theo quy luật phân phối nhị thức với $n=10$ và $p = \mathrm{P}(A \cap B) = 0{,}3$. \\
			Xác suất để có đúng $3$ người sử dụng cả hai loại dịch vụ là
			\[\mathrm{P}(X=3) = \mathrm{C}_{10}^3 \cdot (0{,}3)^3 \cdot (0{,}7)^7 \approx 0{,}2668.\]
			Làm tròn đến hàng phần trăm, ta được $0{,}27$.
			\itemch Gọi $H$ là biến cố \lq\lq Khách hàng được hoàn tiền\rq\rq và $G_1 = A \cap B$ là biến cố khách hàng dùng cả hai dịch vụ. \\
			Ta có $\mathrm{P}(G_1) = 0{,}3 \Rightarrow \mathrm{P}(\overline{G_1}) = 1 - 0{,}3 = 0{,}7$. \\
			Tỷ lệ khách hàng được hoàn tiền nếu sử dụng cả hai dịch vụ: $\mathrm{P}(H \mid G_1) = 0{,}6$. \\
			Tỷ lệ khách hàng được hoàn tiền nếu không sử dụng đồng thời cả hai dịch vụ: $\mathrm{P}(H \mid \overline{G_1}) = 0{,}1$. \\
			Áp dụng công thức xác suất toàn phần, xác suất khách hàng được hoàn tiền là
			\[\mathrm{P}(H) = \mathrm{P}(G_1) \cdot \mathrm{P}(H \mid G_1) + \mathrm{P}(\overline{G_1}) \cdot \mathrm{P}(H \mid \overline{G_1}) = 0{,}3 \cdot 0{,}6 + 0{,}7 \cdot 0{,}1 = 0{,}25.\]
			Xác suất khách hàng thuộc nhóm $G_1$ khi biết người đó được hoàn tiền là
			\[\mathrm{P}(G_1 \mid H) = \dfrac{\mathrm{P}(G_1) \cdot \mathrm{P}(H \mid G_1)}{\mathrm{P}(H)} = \dfrac{0{,}3 \cdot 0{,}6}{0{,}25} = 0{,}72.\]
		\end{itemchoice}
	}
\end{ex}

\begin{ex}%[2H5C3-4]%[ThanhTa]%[26-2-SoGD-VinhLong-L1]

Điểm $A$ thay đổi trên một khung thép hình tròn là giao tuyến của mặt phẳng $(\alpha) \colon x+y-z+4=0$ và mặt cầu $(S) \colon (x-1)^2+(y-1)^2+(z-2)^2=6$. Đèn phát ra luồng sáng có dạng hình nón với góc ở đỉnh bằng $60^{\circ}$, trục của hình nón luôn vuông góc với mặt sàn $(P) \colon x-2y+2z+10=0$. Để đảm bảo mật độ ánh sáng tập trung cao nhất cho khu vực trưng bày dưới sàn, kiến trúc sư cần điều chỉnh đèn đến vị trí sao cho diện tích vùng chiếu sáng trên mặt sàn $(P)$ là nhỏ nhất. Hãy tính diện tích vùng chiếu sáng nhỏ nhất đó. (Không làm tròn các kết quả trung gian, chỉ làm tròn kết quả cuối cùng đến hàng phần mười của $\mathrm{m}^2$).
	\par\shortans{26,2}
	\loigiai{
		\textbf{Bước 1: Xác định đường tròn giao tuyến $(C)$} \\
		Mặt cầu $(S)$ có tâm $I(1;1;2)$ và bán kính $R=\sqrt{6}$. \\
		Khoảng cách từ tâm $I$ đến mặt phẳng quỹ đạo $(\alpha) \colon x+y-z+4=0$ là
		\[\mathrm{d}(I,(\alpha)) = \dfrac{|1+1-2+4|}{\sqrt{1^2+1^2+(-1)^2}} = \dfrac{4}{\sqrt{3}}.\]
		Bán kính của đường tròn khung thép $(r) \colon$
		\[r = \sqrt{R^2 - [\mathrm{d}(I,(\alpha))]^2} = \sqrt{6 - \left(\dfrac{4}{\sqrt{3}}\right)^2} = \sqrt{6 - \dfrac{16}{3}} = \sqrt{\dfrac{2}{3}}.\]
		Tâm $H$ của đường tròn khung thép là hình chiếu của $I$ trên $(\alpha)$. Đường thẳng $IH$ đi qua $I$ và vuông góc với $(\alpha)$ có phương trình tham số $\heva{&x=1+t \\ &y=1+t \\ &z=2-t.}$ \\
		Thay tọa độ $H(1+t;1+t;2-t)$ vào $(\alpha)$ ta được:
		\[(1+t)+(1+t)-(2-t)+4=0 \Rightarrow 3t+4=0 \Rightarrow t = -\dfrac{4}{3}.\]
		Suy ra tọa độ tâm $H$ là $H\left(-\dfrac{1}{3}; -\dfrac{1}{3}; \dfrac{10}{3}\right)$. \\
		\textbf{Bước 2: Thiết lập mối liên hệ giữa diện tích vùng sáng và khoảng cách} \\
		Vì trục hình nón luôn vuông góc với mặt sàn $(P)$, bán kính vùng sáng $R_s$ trên sàn phụ thuộc vào khoảng cách $h$ từ đèn $A$ đến $(P)$ và nửa góc ở đỉnh ($\beta = 30^{\circ}$): $R_s = h \cdot \tan 30^{\circ} = \dfrac{h}{\sqrt{3}}$. \\
		Diện tích vùng chiếu sáng: $S = \pi \cdot R_s^2 = \dfrac{\pi \cdot h^2}{3}$. \\
		Để $S$ nhỏ nhất, ta cần tìm giá trị nhỏ nhất của khoảng cách $h = \mathrm{d}(A,(P))$ với $A$ nằm trên đường tròn tâm $H$, bán kính $r$. \\
		\textbf{Bước 3: Tìm giá trị nhỏ nhất của khoảng cách $h$} \\
		Khoảng cách từ tâm $H$ đến mặt sàn $(P) \colon x-2y+2z+10=0$ là
		\[\mathrm{d}(H,(P)) = \dfrac{\left| -\dfrac{1}{3} - 2 \cdot \left(-\dfrac{1}{3}\right) + 2 \cdot \left(\dfrac{10}{3}\right) + 10 \right|}{\sqrt{1^2 + (-2)^2 + 2^2}} = \dfrac{17}{3}.\]
		Gọi $\varphi$ là góc giữa mặt phẳng khung thép $(\alpha)$ và mặt sàn $(P)$. Ta có
		\[\cos \varphi = \dfrac{|\vec{n}_{\alpha} \cdot \vec{n}_P|}{|\vec{n}_{\alpha}| \cdot |\vec{n}_P|} = \dfrac{|1 \cdot 1 + 1 \cdot (-2) + (-1) \cdot 2|}{\sqrt{3} \cdot 3} = \dfrac{1}{\sqrt{3}} \Rightarrow \sin \varphi = \sqrt{\dfrac{2}{3}}.\]
		Khoảng cách nhỏ nhất từ $A$ đến $(P)$ được xác định bởi:
		\[h_{\min} = \mathrm{d}(H,(P)) - r \cdot \sin \varphi = \dfrac{17}{3} - \sqrt{\dfrac{2}{3}} \cdot \sqrt{\dfrac{2}{3}} = \dfrac{17}{3} - \dfrac{2}{3} = 5\,\mathrm{m}.\]
		\textbf{Bước 4: Tính diện tích nhỏ nhất} \\
		Bán kính vùng được chiếu sáng nhỏ nhất trên mặt sàn là $\min R_s = \dfrac{5}{\sqrt{3}}$. \\
		Giá trị diện tích nhỏ nhất là $S_{\min} = \dfrac{\pi \cdot 5^2}{3} = \dfrac{25\pi}{3} \approx 26,1799 \ldots$ \\
		Làm tròn đến hàng phần mười theo yêu cầu đề bài: $S_{\min} \approx 26,2\,\mathrm{m}^2$.
	}

		\textbf{Bước 1: Viết phương trình đường tròn và elip} \\
		Chọn hệ trục tọa độ $Oxy$ có gốc $O$ trùng với giao điểm của hai trục lớn của hai elip. Khi đó:

			\item Phương trình elip $(E_1)$ có trục lớn nằm trên trục $Ox$ ($a=4, b=\dfrac{4}{\sqrt{3}}$) là 
			\[\dfrac{x^2}{16} + \dfrac{3y^2}{16} = 1 \Rightarrow y^2 = \dfrac{16-x^2}{3} \Rightarrow y = \sqrt{\dfrac{16-x^2}{3}} \text{ (với } y \ge 0).\]

\begin{ex}%[1D2C3-8]%[ThanhTa]%[26-2-SoGD-VinhLong-L1]
	Bác Minh mua một chiếc ti vi tại một cửa hàng với giá $21{,}5$ triệu đồng và đã trả trước $10$ triệu đồng ngay khi nhận ti vi. Số tiền còn lại bác lựa chọn trả góp trong vòng $12$ tháng với lãi suất $3{,}5\%$/tháng theo hình thức lãi kép trên dư nợ giảm dần, tiền lãi được tính dựa trên số dư nợ thực tế tại thời điểm tính lãi. Biết rằng, vào cuối mỗi tháng, bác Minh phải trả cho cửa hàng một số tiền không đổi là $m$ triệu đồng để sau đúng $12$ tháng thì hết nợ. Giá trị của $m$ bằng bao nhiêu? (Không làm tròn các kết quả trung gian, chỉ làm tròn kết quả cuối cùng đến hàng phần trăm).
	\par\shortans{1,19}
	\loigiai{
		Đặt $r = 3{,}5\%$ là lãi suất hàng tháng. \\
		Gọi số tiền bác Minh phải trả góp cho cửa hàng mỗi tháng là $m$ (triệu đồng). \\
		Số tiền vay ban đầu là $A = 21{,}5 - 10 = 11{,}5$ (triệu đồng).
		\begin{itemize}
			\item Số tiền bác Minh còn nợ sau tháng thứ 1: $T_1 = A + Ar - m = A(1+r) - m$.
			\item Số tiền bác Minh còn nợ sau tháng thứ 2:
			\[T_2 = T_1 + T_1r - m = T_1(1+r) - m = A(1+r)^2 - m(1+r+1).\]
			\item Số tiền bác Minh còn nợ sau tháng thứ 3:
			\[T_3 = T_2 + T_2r - m = T_2(1+r) - m = A(1+r)^3 - m((1+r)^2 + (1+r) + 1).\]
			\item Số tiền bác Minh còn nợ sau tháng thứ 12:
			\begin{eqnarray*}
				T_{12} &=& T_{11} + T_{11}r - m = T_{11}(1+r) - m \\
				&=& A(1+r)^{12} - m((1+r)^{11} + (1+r)^{10} + \cdots + (1+r) + 1) \\
				&=& A(1+r)^{12} - m \dfrac{(1+r)^{12}-1}{(1+r)-1}.
			\end{eqnarray*}
		\end{itemize}
		Bác Minh trả đúng $12$ tháng thì hết nợ nên:
		\[T_{12} = 0 \Leftrightarrow m = \dfrac{A(1+r)^{12} \cdot r}{(1+r)^{12}-1}.\]
		Thay số: $m = \dfrac{11{,}5 \cdot (1 + 3{,}5\%)^{12} \cdot 3{,}5\%}{(1 + 3{,}5\%)^{12}-1} \approx 1{,}19$ (triệu đồng).
	}
\end{ex}

\begin{ex}%[2D1H3-6]%[ThanhTa]%[26-2-SoGD-VinhLong-L1]
	Một nghệ nhân làm gốm thủ công tại địa phương đang sản xuất một dòng bình hoa gốm đặc biệt. Mỗi ngày, nghệ nhân này có thể sản xuất $x$ chiếc bình hoa ($1 \le x \le 18, x \in \mathbb{N}$). Tổng chi phí sản xuất $x$ bình hoa trong một ngày (bao gồm nguyên liệu, công thợ và lò nung), tính bằng đơn vị nghìn đồng, được xác định bởi hàm chi phí: $C(x) = x^3 - 3x^2 + 40x + 500$. Giả sử toàn bộ số bình hoa sản xuất ra trong ngày đều được các cửa hàng lưu niệm thu mua hết với giá cố định là $280$ nghìn đồng/bình. Để đạt được lợi nhuận cao nhất, nghệ nhân này nên sản xuất bao nhiêu bình hoa mỗi ngày?
	\par\shortans{10}
	\loigiai{
		Khi bán $x$ bình hoa:
		\begin{itemize}
			\item Số tiền thu được là: $B(x) = 280x$ (nghìn đồng).
			\item Lợi nhuận thu được là: $L(x) = B(x) - C(x) = -x^3 + 3x^2 + 240x - 500$ (nghìn đồng).
		\end{itemize}
		Xét hàm số $L(x)$ trên $[1; 18]$ (với $x$ là số nguyên), ta có:
		\[L'(x) = -3x^2 + 6x + 240.\]
		Cho $L'(x) = 0 \Leftrightarrow \hoac{&x = 10 \text{ (nhận)} \\ &x = -8 \text{ (loại).}}$
		Ta tính giá trị của hàm số tại các điểm đầu mút và điểm cực trị:
		\begin{itemize}
			\item $L(1) = -258$.
			\item $L(10) = 1200$.
			\item $L(18) = -1040$.
		\end{itemize}
		Suy ra tại $x = 10$ thì hàm số đạt giá trị lớn nhất là $1200$. \\
		Như vậy, nghệ nhân nên sản xuất mỗi ngày $10$ bình hoa để thu được lợi nhuận cao nhất.
	}
\end{ex}

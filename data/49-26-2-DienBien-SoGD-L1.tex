\begin{ex}%[2D6H2-4]
	Trong một kì thi tốt nghiệp trung học phổ thông, một tỉnh $X$ có $50\%$ học sinh lựa chọn tổ hợp $A00$ (gồm các môn Toán, Lý, Hóa). Biết rằng, nếu một học sinh chọn tổ hợp $A00$ thì xác suất để học sinh đó đỗ đại học là $0{,}7$; còn nếu một học sinh không chọn tổ hợp $A00$ thì xác suất để học sinh đó đỗ đại học là $0{,}6$. Chọn ngẫu nhiên một học sinh của tỉnh $X$ đã tốt nghiệp trung học phổ thông trong kì thi trên. Biết rằng học sinh này đã đỗ đại học. Gọi 
	\begin{itemize}
	\item $A$ là biến cố: \lq\lq Học sinh đó chọn tổ hợp $A00$\rq\rq.
	\item $B$ là biến cố: \lq\lq Học sinh đó đỗ đại học\rq\rq.
	\end{itemize}
	Khi đó
	\choiceTF
	{\True Xác suất $\mathrm{P}\left(\overline{A}\right)=0{,}5$}
	{Xác suất $\mathrm{P}(B\mid A)=0{,}4$}
	{\True Xác suất $\mathrm{P}\left(B\mid \overline{A}\right)$ thuộc khoảng $(0{,}4; 0{,}9)$}
	{$\dfrac{\mathrm{P}(A\mid B)}{\mathrm{P}(B\mid A)}$ lớn hơn $1$}
	\loigiai{
		Theo giả thiết: $\mathrm{P}(A) = 0{,}5 \Rightarrow \mathrm{P}\left(\overline{A}\right) = 1 - 0{,}5 = 0{,}5$.\\
		Xác suất đỗ đại học khi chọn $A00$: $\mathrm{P}(B\mid A) = 0{,}7$.\\
		Xác suất đỗ đại học khi không chọn $A00$: $\mathrm{P}\left(B\mid \overline{A}\right) = 0{,}6$.
		\begin{itemchoice}
			\itemch Vì $\mathrm{P}\left(\overline{A}\right) = 0{,}5$.
			\itemch $\mathrm{P}(B\mid A) = 0{,}7$.
			\itemch $\mathrm{P}\left(B\mid \overline{A}\right) = 0{,}6 \in (0{,}4; 0{,}9)$.
			\itemch Ta có $\mathrm{P}(B) = \mathrm{P}(B\mid A)\mathrm{P}(A) + \mathrm{P}\left(B\mid \overline{A}\right)\mathrm{P}\left(\overline{A}\right) = 0{,}7 \cdot 0{,}5 + 0{,}6 \cdot 0{,}5 = 0{,}65$.\\
			Ta có $\mathrm{P}(A\mid B) = \dfrac{\mathrm{P}(A \cap B)}{\mathrm{P}(B)} = \dfrac{0{,}7 \cdot 0{,}5}{0{,}65} = \dfrac{7}{13}$.\\
			Tỉ số $\dfrac{\mathrm{P}(A\mid B)}{\mathrm{P}(B\mid A)} = \dfrac{7/13}{7/10} = \dfrac{10}{13} < 1$.
		\end{itemchoice}
	}
\end{ex}

\begin{ex}%[2D4V2-6]
	Một kỹ sư cơ khí nghiên cứu quá trình làm nguội một chi tiết máy sau khi đúc. Độ chênh lệch nhiệt độ giữa vật và môi trường thay đổi theo thời gian. Gọi $y(t)$ là hiệu số nhiệt độ giữa chi tiết máy và môi trường tại thời điểm $t$ (phút). Theo định luật Newton về làm lạnh, tốc độ thay đổi của $y(t)$ tỉ lệ thuận với $y(t)$, tức là $y'(t)=k \cdot y(t)$ $(t\ge0,k<0)$. Người ta đo được nhiệt độ chênh lệch tại thời điểm $t=10$ phút là $40^\circ$C. Đến thời điểm $t=20$ phút, nhiệt độ chênh lệch giảm xuống còn $10^\circ$C. Cho biết $y(t)=\mathrm{e}^{g(t)}$.
	\choiceTF
	{\True Hàm số $g(t)=kt+C$ $(t\ge0)$ với $C$ là một hằng số xác định}
	{\True Giá trị của hằng số $k=-\dfrac{\ln 4}{10}$}
	{Hằng số $C$ (trong biểu thức $g(t)=kt+C$) có giá trị bằng $3\ln 2+\ln 5$}
	{Để nhiệt độ chênh lệch $y(t)$ nhỏ hơn $1^\circ$C thì cần ít nhất $35$ phút kể từ lúc bắt đầu theo dõi}
	\loigiai{
		\begin{itemchoice}
			\itemch Ta có $y'(t) = k \cdot y(t)\Rightarrow \dfrac{y'(t)}{y(t)}=k$ suy ra $\ln y(t) = kt + C\Rightarrow y(t) = \mathrm{e}^{kt + C}$. \\
			Suy ra $g(t) = kt + C$.
			\itemch Ta có $y(10) = 40 \Rightarrow \mathrm{e}^{10k + C} = 40$ và $y(20) = 10 \Rightarrow \mathrm{e}^{20k + C} = 10$.\\
			Khi đó $\dfrac{\mathrm{e}^{20k+C}}{\mathrm{e}^{10k+C}} = \dfrac{10}{40} \Rightarrow \mathrm{e}^{10k} = \dfrac{1}{4} \Rightarrow 10k = -\ln 4 \Rightarrow k = -\dfrac{\ln 4}{10}$.
			\itemch  Với $k = -\dfrac{\ln 4}{10}$ thì $\mathrm{e}^{-\ln 4 + C} = 40 \Rightarrow \dfrac{1}{4} \cdot \mathrm{e}^{C} = 40 \Rightarrow \mathrm{e}^{C} = 160 \Rightarrow C = \ln 160$.\\
			Ta có $C = \ln 160 = \ln(2^5 \cdot 5) = 5\ln 2 + \ln 5 \ne 3\ln 2 + \ln 5$.
			\itemch Ta có $y(t) < 1 \Leftrightarrow 160 \cdot (0{,}25)^{\tfrac{t}{10}} < 1 \Leftrightarrow 4^{\tfrac{t}{10}} > 160 \Leftrightarrow \dfrac{t}{10} > \log_4 160 \Leftrightarrow t > 10 \log_4 160 \approx 36{,}6$.\\
			Vậy cần ít nhất khoảng $37$ phút.
		\end{itemchoice}
	}
\end{ex}

\begin{ex}%[0D0C2-7]
	Trong một trò chơi xếp số, người ta chọn ngẫu nhiên $5$ số khác nhau từ tập $S=\{1;2;3;...;30\}$ và đặt mỗi số vào đúng một vị trí trong $5$ vị trí $A$, $B$, $C$, $D$, $E$ như hình vẽ, sao cho bên trong mỗi vị trí chỉ được xếp một số. Trò chơi kết thúc khi các điều kiện sau được thỏa mãn:
	\begin{itemize}
		\item  Những bộ ba vị trí $(A, B, C)$, $(C, D, E)$ theo thứ tự lập thành cấp số cộng.
		\item  Bộ ba vị trí $(A, C, E)$ theo thứ tự lập thành cấp số nhân.
	\end{itemize}
	Xác suất kết thúc trò chơi ở một lần chọn và xắp sếp là $\dfrac{1}{2\,026a}$. Tính $a$ (Kết quả làm tròn đến chữ số hàng đơn vị).
	\begin{center}
		\begin{tabular}{|c|c|c|c|c|}
			\hline
			$A$	& $B$ & $C$ & $D$ & $E$ \\
			\hline
		\end{tabular}
	\end{center}
	\shortans{469}
	\loigiai{
		Không gian mẫu $n(\Omega) = \mathrm{A}_{30}^5 = 17\,100\,720$.\\
		Ta cần tìm số bộ $(A,B,C,D,E)$ thỏa mãn
		\begin{itemize}
		\item $A+C = 2B$ $(1)$.
		\item $C+E = 2D$ $(2)$.
		\item $A \cdot E = C^2$ $(3)$ (với $A \ne C \ne E$).
		\end{itemize}
		Từ $(1)$ và $(2)$ suy ra $A$ và $C$ phải cùng tính chẵn lẻ, $C$ và $E$ phải cùng tính chẵn lẻ. \\
		Suy ra $A$, $C$, $E$ phải có cùng tính chẵn lẻ (cùng chẵn hoặc cùng lẻ). \\
		Nếu $A$, $C$, $E$ lập thành cấp số nhân thì công bội $q = \dfrac{m}{n}$ $(m,n \in \mathbb{N}^*, (m,n)=1, m > n)$.\\
		Ta có $A = k n^2$, $C = k\cdot m\cdot n$, $E = k m^2$.\\
		Và ngược lại nếu dãy giảm (do tính đối xứng, ta chỉ cần tìm dãy tăng rồi nhân đôi).\\
		Điều kiện $E \le 30 \Rightarrow k m^2 \le 30$.
		\begin{itemize}
			\item Xét các dãy $(A,C,E)$ tăng:\\
			Nếu $m=2$, $n=1$ suy ra $A=k$, $C=2k$, $E=4k$. \\
			Để $A$, $C$, $E$ cùng chẵn lẻ thì $k$ phải chẵn ($k=2p$). \\
			$E = 8p \le 30 \Rightarrow p \in \{1, 2, 3\}$.\\
			Có $3$ bộ: $(2,4,8)$, $(4,8,16)$, $(6,12,24)$.
			\item  Nếu $m=3$, $n=1$ suy ra $A=k$, $C=3k$, $E=9k$.\\
			$E = 9k \le 30 \Rightarrow k \in \{1, 2, 3\}$.\\
			Với $k=1 \Rightarrow (1,3,9)$ (cùng lẻ).\\
			Với $k=2 \Rightarrow (2,6,18)$ (cùng chẵn).\\
			Với $k=3 \Rightarrow (3,9,27)$ (cùng lẻ).\\
			Vậy có $3$ bộ.
			\item  Nếu $m=3$, $n=2$ suy ra $A=4k$, $C=6k$, $E=9k$.\\
			Để cùng chẵn lẻ, $k$ phải chẵn ($k=2p$).\\
			$E = 18p \le 30 \Rightarrow p=1 \Rightarrow k=2$.\\
			Có $1$ bộ: $(8,12,18)$ (cùng chẵn).
			\item  Nếu $m=5$, $n=1$ suy ra $A=k$, $C=5k$, $E=25k$.\\
			$E \le 30 \Rightarrow k=1$.\\
			Có $1$ bộ: $(1,5,25)$ (cùng lẻ).
			\item  Nếu $m=5$, $n=3$ suy ra $A=9k$, $C=15k$, $E=25k$.\\
			$E \le 30 \Rightarrow k=1$.\\
			Có $1$ bộ: $(9,15,25)$ (cùng lẻ).
		\end{itemize}
		Tổng cộng số bộ $(A,C,E)$ tăng thỏa mãn là $3 + 3 + 1 + 1 + 1 = 9$ bộ.\\
		Vì $B$ và $D$ là trung bình cộng nên sẽ được xác định duy nhất từ mỗi bộ $(A,C,E)$. Dễ thấy các giá trị sinh ra luôn phân biệt và nhỏ hơn $30$.\\
		Do dãy có thể giảm dần, tổng số cách xếp là $9 \times 2 = 18$ cách.\\
		Xác suất $P = \dfrac{18}{17\,100\,720} = \dfrac{1}{950\,040}$.\\
		Theo bài ra: $P = \dfrac{1}{2\,026 a} \Rightarrow a = \dfrac{950\,040}{2\,026} \approx 469$.
	}

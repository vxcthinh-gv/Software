\begin{ex}%[2D6H2-4]

Trong số các bài đăng bị đánh dấu, có $12\%$ bài đăng đánh dấu nhầm (bài đăng không vi phạm Luật an ninh mạng nhưng bị đánh dấu). Trong số các bài đăng không bị đánh dấu, vẫn có $6\%$ bài đăng vi phạm Luật an ninh mạng. Chọn ngẫu nhiên một bài đăng trên trang mạng xã hội này.
	\choiceTF
	{\True Xác suất chọn được bài đăng không bị hệ thống đánh dấu bằng $0{,}75$}
	{Xác suất chọn được bài đăng vi phạm, biết rằng nó không bị đánh dấu bằng $0{,}12$}
	{\True Xác suất chọn được bài đăng vi phạm bằng $0{,}265$}
	{\True Xác suất chọn được bài đăng bị đánh dấu, biết rằng nó là bài đăng vi phạm xấp xỉ bằng $0{,}83$}
	\loigiai{
		Gọi biến cố $A$ là "chọn được bài đăng bị đánh dấu" và biến cố $B$ là "chọn được bài đăng vi phạm".
		\begin{itemchoice}
			\itemch $\mathrm{P}(\overline{A})=1-\mathrm{P}(A)=1-0{,}25=0{,}75$.
			\itemch $\mathrm{P}(B|\overline{A})=6\%=0{,}06$.
			\itemch $\mathrm{P}(B)=\mathrm{P}(A)\cdot \mathrm{P}(B|A)+\mathrm{P}(\overline{A})\cdot \mathrm{P}(B|\overline{A})=0{,}25\cdot 0{,}88+0{,}75\cdot 0{,}06=0{,}265$.
			\itemch $\mathrm{P}(A|B)=\dfrac{\mathrm{P}(AB)}{\mathrm{P}(B)}=\dfrac{0{,}25\cdot 0{,}88}{0{,}265}=\dfrac{44}{53}\approx 0{,}83$.
		\end{itemchoice}
	}
\end{ex}

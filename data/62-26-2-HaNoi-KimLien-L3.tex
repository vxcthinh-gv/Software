\begin{ex}%[2H2H1-2]

	\choice
	{\True $k=2$}
	{$k=3$}
	{$k=\dfrac{1}{3}$}

	\loigiai{

		\begin{align*}
			\vec{AC} + \vec{BD} &= (\vec{AM} + \vec{MN} + \vec{NC}) + (\vec{BM} + \vec{MN} + \vec{ND}) \\
			&= 2\vec{MN} + (\vec{AM} + \vec{BM}) + (\vec{NC} + \vec{ND}).
		\end{align*}

		Suy ra $\vec{AC} + \vec{BD} = 2\vec{MN}$. Vậy $k=2$.
	}
